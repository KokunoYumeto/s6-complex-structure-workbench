\end{align}

\end{equation}

\begin{definition}
\label{def:valid-cusp-cutoff}
Let $\Delta\Subset\mathcal A$ be a relatively compact parameter disc.  A
number $\varepsilon>0$ is called a \emph{valid cusp cutoff for $\Delta$} if
there exists a number $\varepsilon_1$ such that
\begin{enumerate}[label=\textup{(V\arabic*)}]
\item
\[
  0<\varepsilon<\varepsilon_1<\min\{1,\varepsilon_0\};
\]
\item with
\[
 M_C(\Delta,\varepsilon_1)
 =\sup_{\substack{|q|\le\varepsilon_1\\a\in\overline\Delta}}
   \bigl\|-2\pi\Im C_a(q)\bigr\|,
\]
one has
\begin{equation}
\label{eq:valid-cutoff-inequality}
 \frac{\|B_0^{-1}\|M_C(\Delta,\varepsilon_1)}
      {|\log\varepsilon|}\le\frac12;
\end{equation}
\item the closed subdisc $\overline D_\varepsilon=\{|q|\le\varepsilon\}$,
      viewed inside the original cusp coordinate disc
      $D_{\varepsilon_0}\cup\{p_0\}\subset B$, is disjoint from
      $\overline D_1\cup\overline D_2$.
\end{enumerate}
A number $\varepsilon_1$ satisfying these conditions is called an
\emph{auxiliary cutoff} for $\varepsilon$.
\end{definition}
