\begin{cor}\label{cor: pullback to product central fibre at 0}
There is a $\lambda\in\IC$ with $\operatorname{Im}\lambda>0$ such that
$ \tilde X_0\isom \hat X_0 /\langle\iota\rangle$ where 
\[
 \hat  X_0=
 \left(
 \frac{\IC}{2\IZ+(1+i)\IZ}
 \times
 \frac{\IC}{\IZ+\lambda\IZ}
 \right) \text { and }
 \iota(c,w)=\left(c+1,w+\frac12\right).
\]
In particular, $\tilde X_0$ is an abelian surface.
\end{cor}
