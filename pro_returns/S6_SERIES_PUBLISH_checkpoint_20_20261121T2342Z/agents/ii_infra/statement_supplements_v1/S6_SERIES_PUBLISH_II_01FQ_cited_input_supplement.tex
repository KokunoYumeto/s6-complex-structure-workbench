\begin{proposition}[Credited re-derivation: Homology of all original affine twists]\label{cdp:st:E19}
For the original multiplicities $(3,4)$ and every admissible affine
datum, put $p=12l_0-4r_1-3r_2$ as in Paper~I, Proposition~4.5 and $d=|p|$.
Then $p\ne0$ and
\[
 \pi_1(X)=\mathbb Z/d\mathbb Z,
 \qquad
 H_j(X,\mathbb Z)=
 \begin{cases}
  \mathbb Z&j=0,6,\\
  \mathbb Z/d\mathbb Z&j=1,2,3,4,\\
  0&j=5.
 \end{cases}
\]
In particular every $|p|>1$ member is a rational homology
six-sphere, has algebraic dimension one, and has the deformation
and tangent-cohomology conclusions of \ref{cdp:st:E11}.
\end{proposition}

\begin{proposition}[Credited re-derivation]\label{cdp:st:C.1.2}
Let $Y$ be as in Statement \ref{cdp:st:C.1.1} and let $D$ be any nonzero effective
Cartier divisor, with arbitrary positive multiplicities. For every
holomorphic vector bundle $E$ and every component $P$ of $\Pic(Y)$ there
is a dense open subset $P_E\subset P$ on which
\[
 H^0(Y,E\otimes L)=H^3(Y,E\otimes L)=0.
\]
Each excluded locus is a proper closed complex analytic subset of $P$.
\end{proposition}

\begin{proposition}[All coprime $(3k,4l)$ multiplicities]\label{cdp:st:E11}
Fix admissible logarithmic-transformation data with multiplicities
$r=3k$, $s=4l$, where $k,l>0$ and $\gcd(r,s)=1$, in the family
constructed from the original elliptic-surface data. For every member
which is diffeomorphic to $S^6$, as in
\cite[Theorem labelled \texttt{t:main}]{Viaclovsky2026Log},
\[
 h^\bullet(X,T_X)=(1,1,1,0),
\]
its Kuranishi germ is a smooth reduced line, and varying the nonzero
linearization scale with those data fixed is locally miniversal.
For the original $(3,4)$ affine-twist family, the same statements
hold whenever $p\ne0$, including $|p|>1$.
\end{proposition}

\begin{proposition}\label{cdp:st:C.1.1}
Let $Y$ be a connected compact complex threefold with
$b_1(Y)=b_2(Y)=0$. Every holomorphic line bundle $L$ on $Y$ satisfies
\[
 \chi(Y,T_Y\otimes L)=\frac12\int_Yc_3(T_Y)
 =1-\frac{b_3(Y)}2.
\]
The orientation in the integral is the complex orientation. The group
$\Pic(Y)$ has finitely many connected components, each a torsor for the
complex vector group $H^1(Y,\OO_Y)$, and
$\pi_0\Pic(Y)\simeq H^2(Y;\Z)$ through the first Chern class.
\end{proposition}
