\begin{lem}\label{lem: product at 1}
Over a sufficiently small disc $\Delta_1$ centred at $1$, there is
a homeomorphism
\[
 Y_{\Delta_1}\cong S_{\Delta_1}\times S^1\times S^1
\]
over $\Delta_1$. Under this identification, $\delta$ is the
positively oriented first circle, and $\xi$ is induced by
projection to the second circle.
\end{lem}
