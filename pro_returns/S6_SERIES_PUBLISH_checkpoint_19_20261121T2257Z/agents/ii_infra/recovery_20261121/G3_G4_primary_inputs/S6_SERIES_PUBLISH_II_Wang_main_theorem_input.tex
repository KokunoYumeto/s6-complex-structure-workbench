\begin{theorem}\label{main:theorem}
For every sufficiently small nonzero \(\lambda\), the threefold
\(X_2(\lambda)\) is compact, smooth, and simply connected, and
\begin{equation}
 H_k(X_2(\lambda);\Z)\simeq
 \begin{cases}
 \Z,&k=0,6,\\
 \Z/2,&k=2,3,\\
 0,&k=1,4,5.
 \end{cases}
 \label{main:homology}
\end{equation}
In particular, it is a complex rational homology six-sphere.
\end{theorem}
