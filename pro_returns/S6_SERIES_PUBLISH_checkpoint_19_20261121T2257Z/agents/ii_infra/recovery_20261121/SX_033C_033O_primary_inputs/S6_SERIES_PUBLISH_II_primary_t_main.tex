\begin{theorem} 
\label{t:main}
Let $f_0:X_0\to\PP^1$ be a fibration arising from the Alp\"oge--Claude $\mathrm{IV}^*\,\mathrm{III}\,\mathrm{I}_1$ configuration \cite{AC,Engel}. For every integer $m\geq12$,  there is a holomorphic fibration
$f_m:X_m\to\PP^1$ such that
\begin{enumerate}[label=\textup{(\roman*)}]
\item There is a diffeomorphism $\Phi_m:X_0\to X_m$ with
$f_m\circ\Phi_m=f_0$. The integral monodromy, critical values,
multiplicities, and reduced critical fibers are unchanged.
\item The fibration $f_m$ is the algebraic reduction of $X_m$; in particular, $a(X_m)=1$. The Kodaira dimension $\kappa(X_m) = - \infty$.
\item $X_m$ satisfies
\begin{align}\label{newhodge}
 h^{0,1}(X_m)=2m+1,\quad h^{0,2}(X_m)=2m,\quad h^{0,3}(X_m)=0.
\end{align}
\item The local Kuranishi dimension of $X_m$ satisfies
\begin{align}\label{intromoduli}
 h^1(X_m,\Theta_{X_m})\ \geq\
 \dim\operatorname{Def}(X_m)\ \geq\ 8m -8.
\end{align}
\end{enumerate}
\end{theorem}
