\begin{proof}[Application of the cited Torelli classification]
\begin{enumerate}[label=\arabic*.]
\item Use Jin's oriented $S^2\times S^3$, its positive $H^2$ generator,
and its seam $(z,0)\sim(f(z),1)$. His mapping-torus map identifies
$f\mapsto(A(f),k(f))$ with an isomorphism to $\mathbb Z^2$ for
$g=1$ \cite[Proposition~3.1, pp.~10--11; Theorem~4.9, p.~24]{Jin2026},
where $A=(4k-\ell)/24$. We use that classification as a cited result.
\item The source twists are
$a=t_{S_1,\alpha}$ and $b=t_{S_1\#S_1,\alpha}$, with the actual
local map $(x,y)\mapsto(\alpha(y)x,y)$ and preferred clutch
$\langle p_1(\xi_\alpha),[S^4]\rangle=4$.
Jin's evaluations give
$(k,\ell)(a)=(1,4)$ and $(k,\ell)(b)=(8,8)$
\cite[Lemma~4.4, pp.~16--17; Lemma~4.8 and the $g=1$ table, p.~23]{Jin2026}.
Thus $b-8a,a$ map to the ordered standard basis in coordinates $(A,k)$.
\item The exact coordinate inverse is therefore
\[
 [f]=(k(f)-8A(f))a+A(f)b. \tag{P3F.3.1}
\]
It also gives $\ell\equiv4k\pmod{24}$. The pair $(-1,-4)$
has $A=0$ and maps to $-a$. Its specified Hopf-gluing image is
computed by the explicit extension in \cref{night:P3F.4}.
\end{enumerate}
\end{proof}
