  \loop
    \expandafter\gdef\csname AuditStatus@\the\AuditPageLoop\endcsname{#3}%
  \ifnum\AuditPageLoop<#2\relax
    \advance\AuditPageLoop by 1\relax
  \repeat
}

\newcommand{\AuditShipoutBadge}{%
  \begingroup
  \edef\AuditPageNumber{\number\value{page}}%
  \ifcsname AuditStatus@\AuditPageNumber\endcsname
    \edef\AuditStatusName{\csname AuditStatus@\AuditPageNumber\endcsname}%
    \edef\AuditStatusLetter{\csname AuditSymbol@\AuditStatusName\endcsname}%
    \edef\AuditStatusColor{\csname AuditColor@\AuditStatusName\endcsname}%
    \raisebox{-13pt}[0pt][0pt]{%
      \makebox[0pt][l]{%
        \hspace*{\dimexpr\paperwidth-31pt\relax}%
        \hyperlink{audit-page-\AuditPageNumber}{%
          \colorbox{\AuditStatusColor}{%
            \strut\color{white}\sffamily\bfseries
            \fontsize{5.2pt}{5.7pt}\selectfont
            \AuditStatusLetter\,\AuditPageNumber
          }%
        }%
      }%
    }%
  \fi
  \endgroup
}
