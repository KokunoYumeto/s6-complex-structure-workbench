00026	\begin{abstract}
00027	We compute, retaining \(\alpha>0\), \(c>0\), \(m_0\), and every singular
00028	orbit type, the presymplectic reduction of
00029	\[
00030	 \int_I\left(\langle\alpha Z^*Z,\dot x\rangle_L
00031	 -\frac e2\bigl(\det(\alpha Z^*Z)-m_0^2c^2\bigr)\right)\dd\tau
00032	\]
00033	by arbitrary \(\tau\)-dependent left \(SU(3)\) transformations.
00034	For \(m_0>0\), the colour action is free and the reduced system is
00035	exactly the future-timelike first-order spinless relativistic particle;
00036	its physical phase space is \(T^*\mathcal H^+_{m_0,c}\).
00037	For \(m_0=0\), the nonzero rank-one physical quotient is
00038	\(T^*\mathcal N_+^\times\), so its unparameterized null trajectories
00039	agree with the standard massless particle.  The original multiplier
00040	system is nevertheless not exactly standard: the pulled-back
00041	constraint has zero differential on the massless locus, \(e\) drops
00042	out of the rank-one velocity equation, four real transverse directions
00043	are first-order null but not colour gauge, and at \(Z=0\) every
00044	\(x(\tau)\) is stationary.

00048	\section{Objects, constants, and equations}
00049	Let
00050	\[
00051	 \J=\{A\in\operatorname{Mat}_{2\times2}(\mathbb C):A=A^*\},
00052	\qquad
00053	 \langle A,B\rangle_L=
00054	 \frac12\bigl(\operatorname{tr}A\operatorname{tr}B-\operatorname{tr}(AB)\bigr).
00055	\]
00056	Then \(\langle A,A\rangle_L=\det A\), and the pairing is nondegenerate.
00057	Write
00058	\[
00059	 \Psd=\{p\in\J:p\succeq0\},\quad
00060	 \Ppos=\{p\in\J:p\succ0\},\quad
00061	 \Nfuture=\{p\in\Psd:\rankC p=1\}.
00062	\]
00063	Retain
00064	\[
00065	 \V=\operatorname{Mat}_{3\times2}(\mathbb C),\quad
00066	 \alpha>0,\quad c>0,\quad m_0\geq0,
00067	\]
00068	and define
00069	\begin{align}
00070	 P_\alpha(Z)&=\alpha Z^*Z,\label{wpr:eq:P}\\
00071	 D P_{\alpha,Z}(\delta Z)&=
00072	 \alpha(\delta Z^*Z+Z^*\delta Z),\label{wpr:eq:DP}\\
00073	 \Phi_{m_0,\alpha,c}(Z)&=
00074	 \frac12\left(\det P_\alpha(Z)-m_0^2c^2\right).\label{wpr:eq:Phi}
00075	\end{align}
00076	The history configuration space on a retained parameter interval \(I\) is
00077	\[
00078	 \mathfrak C_I=
00079	 C^\infty(I,\J)\times C^\infty(I,\V)\times C^\infty(I,\mathbb R),
00080	\]
00081	with coordinates \((x,Z,e)\).  Because the Lagrangian is first order
00082	in \(x\), contains no \(\dot Z\), and contains no \(\dot e\), the
00083	corresponding instantaneous presymplectic phase carrier is
00084	\[
00085	 \mathcal E={\J}_x\times{\V}_Z\times\mathbb R_e.
00086	\]
00087	Define
00088	\begin{equation}
00089	 \theta=\langle P_\alpha(Z),\dd x\rangle_L,\qquad
00090	 \Omega=-\dd\theta,\qquad
00091	 \mathcal H=e\Phi_{m_0,\alpha,c}.
00092	 \label{wpr:eq:forms}
00093	\end{equation}
00094	For tangent vectors \(v=(v_x,v_Z,v_e)\) and \(w=(w_x,w_Z,w_e)\),
00095	\begin{equation}
00096	 \Omega(v,w)=
00097	 \langle v_x,D P_{\alpha,Z}(w_Z)\rangle_L
00098	 -\langle w_x,D P_{\alpha,Z}(v_Z)\rangle_L.
00099	 \label{wpr:eq:Omega}
00100	\end{equation}

00102	\begin{proposition}[Exact presymplectic Euler--Lagrange equation]
00103	\label{wpr:prop:EL}
00104	A path \((x,Z,e)\) is stationary under compactly supported variations
00105	if and only if
00106	\begin{align}
00107	 \det P_\alpha(Z)&=m_0^2c^2,\label{wpr:eq:EL1}\\
00108	 \frac{\dd}{\dd\tau}P_\alpha(Z)&=0,\label{wpr:eq:EL2}\\
00109	 \left\langle D P_{\alpha,Z}(\delta Z),
00110	 \dot x-eP_\alpha(Z)\right\rangle_L&=0
00111	 \quad\text{for every }\delta Z\in\V.\label{wpr:eq:EL3}
00112	\end{align}
00113	Equivalently, \(\iota_{\dot\gamma}\Omega=\dd\mathcal H\).
00114	\end{proposition}
00115	\begin{proof}
00116	Since
00117	\[
00118	 D(\det)_p(A)=2\langle p,A\rangle_L,
00119	\]
00120	one has
00121	\[
00122	 \dd\mathcal H(w)=
00123	 \Phi_{m_0,\alpha,c}w_e+
00124	 e\langle P_\alpha(Z),D P_{\alpha,Z}(w_Z)\rangle_L.
00125	\]
00126	Insert \eqref{wpr:eq:Omega} into
00127	\(\iota_{\dot\gamma}\Omega=\dd\mathcal H\) and test successively
00128	\(w_x,w_e,w_Z\).  This gives \eqref{wpr:eq:EL2},
00129	\eqref{wpr:eq:EL1}, and \eqref{wpr:eq:EL3}.
00130	\end{proof}

00135	\begin{lemma}[Local colour is a zero-moment null symmetry]
00136	\label{wpr:lem:null}
00137	For \(\xi\in\mathfrak{su}(3)\), let \(\xi^\#=(0,\xi Z,0)\).  Then
00138	\[
00139	 D P_{\alpha,Z}(\xi Z)=0,\qquad
00140	 \iota_{\xi^\#}\theta=0,\qquad
00141	 \iota_{\xi^\#}\Omega=0.
00142	\]
00143	Thus the zero map to \(\mathfrak{su}(3)^*\) is a presymplectic moment
00144	map.  The action is invariant under every \(\tau\)-dependent
00145	\(U:I\to SU(3)\), with no connection term.
00146	\end{lemma}
00147	\begin{proof}
00148	Because \(\xi^*=-\xi\),
00149	\[
00150	 D P_{\alpha,Z}(\xi Z)=
00151	 \alpha(Z^*\xi^*Z+Z^*\xi Z)=0.
00152	\]
00153	The form identities follow from \eqref{wpr:eq:forms} and
00154	\eqref{wpr:eq:Omega}.  Pointwise,
00155	\((UZ)^*(UZ)=Z^*Z\), even when \(U\) depends on \(\tau\).
00156	\end{proof}

00158	\begin{theorem}[Gram quotient, fibres, and stabilizers]
00159	\label{wpr:thm:quotient}
00160	The map \(P_\alpha\) induces a homeomorphism
00161	\[
00162	 \V/SU(3)\xrightarrow{\ \sim\ }\Psd,\qquad [Z]\longmapsto P_\alpha(Z).
00163	\]
00164	Its orbit types are
00165	\begin{center}
00166	\begin{tabular}{c c c}
00167	\toprule
00168	\(\rankC Z\)&\text{fibre}&\(\operatorname{Stab}_{SU(3)}(Z)\)\\
00169	\midrule
00170	\(2\)&\(SU(3)\)&\(\{I_3\}\)\\
00171	\(1\)&\(SU(3)/SU(2)\cong S^5\)&\(SU(2)\)\\
00172	\(0\)&a point&\(SU(3)\)\\
00173	\bottomrule
00174	\end{tabular}
00175	\end{center}
00176	The rank-two quotient is smooth and free.  Rank one is a smooth
00177	nonprincipal orbit-type stratum with nontrivial stabilizer; the zero
00178	vertex is the singular apex of the total stratified quotient.
00179	\end{theorem}
00180	\begin{proof}
00181	If \(Z_1^*Z_1=Z_2^*Z_2\), the map \(Z_1v\mapsto Z_2v\) is an isometry
00182	between their images and extends to a unitary map of \(\mathbb C^3\).
00183	Its action on the orthogonal complement can be adjusted to make its
00184	determinant one.  In rank two the complement is one-dimensional and
00185	the adjustment is unique, hence the stabilizer is trivial.  In rank
00186	one the stabilizer is \(SU(2)\) on the two-dimensional complement.
00187	At zero it is all of \(SU(3)\).  Surjectivity has the retained section
00188	\[
00189	 Z_p=\alpha^{-1/2}
00190	 \begin{pmatrix}p^{1/2}\\0\quad0\end{pmatrix}.
00191	\]
00192	Compactness of \(SU(3)\) and continuity of the positive square root
00193	give the quotient homeomorphism.  On \(\Ppos\) the section is smooth.
00194	On \(\Nfuture\), the rank-one identity
00195	\[
00196	 p^{1/2}=\frac{p}{\sqrt{\operatorname{tr}p}}
00197	\]
00198	shows directly that the same section is smooth; together with the
00199	constant orbit type \(SU(3)/SU(2)\), this gives the stated smooth
00200	nonprincipal stratum.
00201	\end{proof}

00203	\begin{lemma}[Exact ranks of the Gram differential]
00204	\label{wpr:lem:ranks}
00205	\begin{enumerate}[label=\textup{(\roman*)}]
00206	\item If \(\rankC Z=2\), then
00207	\[
00208	 \operatorname{im}D P_{\alpha,Z}=\J,\qquad
00209	 \ker D P_{\alpha,Z}=T_Z(SU(3)\cdot Z),
00210	\]
00211	of real dimensions \(4\) and \(8\).
00212	\item If \(\rankC Z=1\) and \(p=P_\alpha(Z)\), then
00213	\[
00214	 \operatorname{im}D P_{\alpha,Z}=p^{\perp_L}=T_p\Nfuture
00215	\]
00216	has real dimension \(3\).  The full ambient kernel has dimension \(9\),
00217	but
00218	\[
00219	 \ker(D P_{\alpha,Z}|_{T_Z{\V}_1})=T_Z(SU(3)\cdot Z)
00220	\]
00221	has dimension \(5\), where \({\V}_1=\{Z:\rankC Z=1\}\).  The remaining
00222	four real kernel directions are
00223	\[
00224	 \operatorname{Hom}_{\mathbb C}
00225	 \bigl(\ker Z,(\operatorname{im}Z)^\perp\bigr);
00226	\]
00227	they raise rank and are not colour gauge.
00228	\item At \(Z=0\), \(D P_{\alpha,0}=0\).
00229	\end{enumerate}
00230	\end{lemma}
00231	\begin{proof}
00232	For rank two and arbitrary \(A=A^*\), retain
00233	\[
00234	 \delta Z_A=\frac1{2\alpha}Z(Z^*Z)^{-1}A.
00235	\]
00236	Substitution into \eqref{wpr:eq:DP} gives
00237	\(D P_{\alpha,Z}(\delta Z_A)=A\).  Hence the kernel has dimension
00238	\(12-4=8\), equal to the free colour-orbit tangent.
00239	
00240	For rank one write, without fixing either norm,
00241	\[
00242	 Z=ab^*,\qquad a\in\mathbb C^3\setminus\{0\},\quad
00243	 b\in\mathbb C^2\setminus\{0\},\quad r=a^*a>0.
00244	\]
00245	Then \(p=\alpha rbb^*\).  If \(w=\delta Z^*a\), every
00246	\(w\in\mathbb C^2\) occurs by taking \(\delta Z=aw^*/r\), and
00247	\[
00248	 \operatorname{im}D P_{\alpha,Z}
00249	 =\{\alpha(wb^*+bw^*):w\in\mathbb C^2\}.
00250	\]
00251	The kernel of \(w\mapsto wb^*+bw^*\) is
00252	\(\{itb:t\in\mathbb R\}\), so the image has dimension three.  It is
00253	contained in \(p^{\perp_L}\), and equality follows by dimension.
00254	
00255	Use the exact orthogonal splittings
00256	\[
00257	 \mathbb C^2=(\ker Z)^\perp\oplus\ker Z,\qquad
00258	 \mathbb C^3=\operatorname{im}Z\oplus(\operatorname{im}Z)^\perp.
00259	\]
00260	Retaining the nonzero singular value \(\sigma>0\), adapted unit vectors
00261	give
00262	\[
00263	 Z=\begin{pmatrix}\sigma&0\\0&0\\0&0\end{pmatrix},\qquad
00264	 \delta Z=\begin{pmatrix}a_0&b_0\\c_1&d_1\\c_2&d_2\end{pmatrix},
00265	\]
00266	and
00267	\[
00268	 \alpha^{-1}D P_{\alpha,Z}(\delta Z)=
00269	 \begin{pmatrix}
00270	 2\sigma\operatorname{Re}a_0&\sigma b_0\\
00271	 \sigma\overline{b_0}&0
00272	 \end{pmatrix}.
00273	\]
00274	The ambient kernel has \(\operatorname{Re}a_0=b_0=0\) and dimension
00275	nine.  The linearized rank-one equations also impose \(d_1=d_2=0\).
00276	The five remaining directions are the \(SU(3)/SU(2)\)-orbit tangent;
00277	the \(d_1,d_2\) block is the displayed complex Hom-space.  The zero
00278	case follows from \eqref{wpr:eq:DP}.
00279	\end{proof}

00290	\begin{proposition}[Reduced one-form and two-form]
00291	\label{wpr:prop:reduced-forms}
00292	The forms are basic for the colour action and satisfy
00293	\[
00294	 \theta=q^*\overline\theta,\qquad
00295	 \Omega=q^*\overline\Omega,
00296	\]
00297	where
00298	\begin{align}
00299	 \overline\theta&=\langle p,\dd x\rangle_L,\label{wpr:eq:bartheta}\\
00300	 \overline\Omega&=-\dd\overline\theta,\label{wpr:eq:barOmega}\\
00301	 \overline\Omega((v_x,v_p),(w_x,w_p))
00302	 &=\langle v_x,w_p\rangle_L-\langle w_x,v_p\rangle_L.
00303	 \label{wpr:eq:barOmega-explicit}
00304	\end{align}
00305	After omitting the multiplier coordinate, this is symplectic on
00306	\(\J\times\Ppos\).  On \(\J\times\Nfuture\), its kernel is exactly
00307	\[
00308	 \ker\overline\Omega_{(x,p)}=\SpanR\{(p,0)\}.
00309	\]
00310	On \(\J\times\{0\}\), the stratum-restricted two-form is zero.
00311	\end{proposition}
00312	\begin{proof}
00313	Basicness follows from Lemma~\ref{wpr:lem:null}; the formulas follow by
00314	differentiation.  The rank-two assertion uses that \(\Ppos\) is open in
00315	\(\J\).  At nonzero rank-one \(p\), one has
00316	\[
00317	 T_p\Nfuture=p^{\perp_L}.
00318	\]
00319	If \((v_x,v_p)\) annihilates \eqref{wpr:eq:barOmega-explicit}, testing
00320	arbitrary \(w_x\) gives \(v_p=0\), and testing
00321	\(w_p\in p^{\perp_L}\) gives
00322	\(v_x\in(p^{\perp_L})^{\perp_L}=\SpanR\{p\}\).
00323	At \(p=0\), the zero stratum has no momentum tangent direction.
00324	\end{proof}

00326	\section{Positive mass}
00327	For \(m_0>0\), define the retained future shell
00328	\[
00329	 \mathcal H^+_{m_0,c}
00330	 =\{p\in\Ppos:\det p=m_0^2c^2\}.
00331	\]
00332	The parameter \(\alpha\) remains present in the exact quotient map
00333	\(p=\alpha Z^*Z\); the displayed subset of momentum space itself
00334	depends on \(m_0,c\).
00335	
00336	\begin{theorem}[Exact massive reduction]
00337	\label{wpr:thm:massive}
00338	Let \(m_0>0\).  Then:
00339	\begin{enumerate}[label=\textup{(\alph*)}]
00340	\item The constraint surface
00341	\[
00342	 \Sigma_{m_0,\alpha,c}
00343	 =\J\times\{Z:\det P_\alpha(Z)=m_0^2c^2\}\times\mathbb R_e
00344	\]
00345	is smooth and the colour action on it is free.
00346	\item Its colour quotient is
00347	\[
00348	 \overline\Sigma_{m_0,c}
00349	 =\J\times\mathcal H^+_{m_0,c}\times\mathbb R_e.
00350	\]
00351	Before colour reduction the kernel of the restricted two-form is the
00352	direct sum of the eight-dimensional colour-orbit tangent,
00353	\(\SpanR\{(p,0)\}\), and \(\SpanR\{\partial_e\}\).  After colour
00354	reduction and omission of \(e\), only \(\SpanR\{(p,0)\}\) remains.
00355	\item The equations are
00356	\[
00357	 \dot p=0,\qquad \dot x=e\,p,\qquad
00358	 \det p=m_0^2c^2,
00359	\]
00360	while \(\dot e\) and motion in the colour fibre are arbitrary.
00361	\item The physical phase space
00362	\[
00363	 \mathcal R_{m_0,c}
00364	 =
00365	 \frac{\J\times\mathcal H^+_{m_0,c}}
00366	 {(x,p)\sim(x+s p,p),\ s\in\mathbb R}
00367	\]
00368	is canonically symplectomorphic to \(T^*\mathcal H^+_{m_0,c}\).
00369	\item The colour-reduced positive-mass model is exactly the standard
00370	future-directed spinless first-order relativistic particle of mass
00371	\(m_0\) and speed constant \(c\).
00372	\end{enumerate}
00373	\end{theorem}
00374	\begin{proof}
00375	The constraint implies \(\rankC Z=2\).  By
00376	Lemma~\ref{wpr:lem:ranks}, \(P_\alpha\) is a submersion there, and
00377	\[
00378	 D(\det)_p=2\langle p,\mathord{\cdot}\rangle_L\neq0.
00379	\]
00380	This proves smoothness.  Theorem~\ref{wpr:thm:quotient} gives freeness
00381	and the colour quotient.  Since
00382	\[
00383	 T_p\mathcal H^+_{m_0,c}=p^{\perp_L},
00384	\]
00385	the proof of Proposition~\ref{wpr:prop:reduced-forms} gives the
00386	characteristic line.  Pullback through the Gram submersion adds exactly
00387	the colour-orbit kernel, and the multiplier adds \(\partial_e\).
00388	
00389	Surjectivity of \(D P_{\alpha,Z}\) turns \eqref{wpr:eq:EL3} into
00390	\(\dot x=e p\); the other equations are \eqref{wpr:eq:EL1} and
00391	\eqref{wpr:eq:EL2}.  A path with constant \(p\) lies in one free Gram
00392	fibre and differs locally from a constant lift by a
00393	\(\tau\)-dependent colour transformation.
00394	
00395	For the physical identification, if
00396	\(A\in T_p\mathcal H^+_{m_0,c}\), define
00397	\[
00398	 \xi_{[x],p}(A)=-\langle x,A\rangle_L.
00399	\]
00400	This is unchanged by \(x\mapsto x+s p\).  Conversely, the annihilator
00401	of \(p^{\perp_L}\) is \(\SpanR\{p\}\), so every cotangent covector has
00402	one and only one class \([x]\).  Hence
00403	\[
00404	 ([x],p)\longmapsto(p,\xi_{[x],p})
00405	\]
00406	is a fibrewise linear bijection to \(T^*\mathcal H^+_{m_0,c}\).
00407	The characteristic \(\mathbb R\)-action is free because every shell
00408	momentum is nonzero.  It is proper: choose any Euclidean norm on
00409	\(\J\).  For compact sets \(K_1,K_2\subset
00410	\J\times\mathcal H^+_{m_0,c}\), the momenta occurring in them have a
00411	positive minimum Euclidean norm \(\delta\); if
00412	\((x,p)\in K_1\) and \((x+s p,p)\in K_2\), boundedness of the two
00413	position projections gives \(|s|\delta\leq C(K_1,K_2)\).
00414	Thus the orbit space is smooth.  The smooth vector-bundle map
00415	\[
00416	 (x,p)\longmapsto
00417	 \bigl(p,\;A\mapsto-\langle x,A\rangle_L\bigr)
00418	\]
00419	is fibrewise onto with kernel \(\SpanR\{p\}\), and therefore descends
00420	with a smooth inverse.  The displayed bijection is consequently a
00421	diffeomorphism.
00422	
00423	The basic one-form on the shell is
00424	\begin{equation}
00425	 \theta_{\mathrm{basic}}=-\langle x,\dd p\rangle_L.
00426	 \label{wpr:eq:basic}
00427	\end{equation}
00428	It is the pullback of the tautological cotangent one-form, and
00429	\begin{equation}
00430	 \overline\theta
00431	 =\dd\langle p,x\rangle_L+\theta_{\mathrm{basic}}.
00432	 \label{wpr:eq:boundary}
00433	\end{equation}
00434	Therefore
00435	\(\overline\Omega=-\dd\theta_{\mathrm{basic}}\), the canonical
00436	cotangent symplectic form under the convention
00437	\(\omega_{\mathrm{can}}=-\dd\lambda_{\mathrm{can}}\).
00438	The original \(\overline\theta\) is not horizontal on the massive
00439	characteristic because \(\overline\theta(p,0)=m_0^2c^2\); the exact
00440	boundary term in \eqref{wpr:eq:boundary} produces the reduced
00441	one-form.
00442	
00443	Finally the reduced action is
00444	\[
00445	 \int_I\left(\langle p,\dot x\rangle_L
00446	 -\frac e2(\det p-m_0^2c^2)\right)\dd\tau.
00447	\]
00448	Because \(\Ppos\) is open in \(\J\), \(p\) has every infinitesimal
00449	Hermitian variation.  This is the standard covariant first-order
00450	spinless particle action on the future-timelike sector, with no
00451	residual internal symplectic factor.
00452	\end{proof}

00454	\begin{proposition}[Exact Hamiltonian gauge quasi-symmetry]
00455	\label{wpr:prop:gauge}
00456	For a smooth real \(\varepsilon(\tau)\), set
00457	\[
00458	 \delta_\varepsilon x=\varepsilon P_\alpha(Z),\qquad
00459	 \delta_\varepsilon Z=0,\qquad
00460	 \delta_\varepsilon e=\dot\varepsilon.
00461	\]
00462	Then
00463	\begin{equation}
00464	 \delta_\varepsilon L
00465	 =
00466	 \frac{\dd}{\dd\tau}
00467	 \left[
00468	 \frac{\varepsilon}{2}
00469	 \bigl(\det P_\alpha(Z)+m_0^2c^2\bigr)
00470	 \right].
00471	 \label{wpr:eq:gauge-boundary}
00472	\end{equation}
00473	For compactly supported \(\varepsilon\) this is an exact gauge
00474	quasi-symmetry, whose phase-space part generates the shell
00475	characteristic where that characteristic is regular.
00476	\end{proposition}
00477	\begin{proof}
00478	Put \(p=P_\alpha(Z)\) and \(q=\det p=\langle p,p\rangle_L\).  Since
00479	\(\dot q=2\langle p,\dot p\rangle_L\),
00480	\begin{align*}
00481	 \delta_\varepsilon L
00482	 &=
00483	 \langle p,\dot\varepsilon p+\varepsilon\dot p\rangle_L
00484	 -\frac{\dot\varepsilon}{2}(q-m_0^2c^2)\\
00485	 &=
00486	 \frac{\dot\varepsilon}{2}(q+m_0^2c^2)
00487	 +\frac{\varepsilon}{2}\dot q,
00488	\end{align*}
00489	which is \eqref{wpr:eq:gauge-boundary}.
00490	\end{proof}

00503	\begin{lemma}[The pulled-back massless constraint is irregular]
00504	\label{wpr:lem:irregular}

00506	\begin{equation}
